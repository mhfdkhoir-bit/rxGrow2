async function fetchStockPrice(kode, idx) { if (!kode) return; const targetInput = document.querySelector(`input[data-field="harga"][data-index="${idx}"]`); const symbol = kode.toUpperCase().replace('.JK', '') + '.JK'; // Menggunakan Cloudflare Worker pribadi kamu agar secepat kilat tanpa antre const workerUrl = `https://saham-proxy.m4hfud-backup.workers.dev/?symbol=${symbol}`; try { if(targetInput) targetInput.placeholder = "Load..."; const res = await fetch(workerUrl, { cache: "no-store" }); if (!res.ok) throw new Error("Gagal mengambil data"); const rawData = await res.json(); if (rawData?.chart?.result?.[0]) { const price = rawData.chart.result[0].meta.regularMarketPrice; if (price) { db.saham[idx].harga = price; if(targetInput) { targetInput.value = isCensored ? '••••••' : price.toLocaleString('id-ID'); targetInput.placeholder = "0"; } saveDB(); updateSahamRowUI(idx); calculateTotals(); return; } } if(targetInput) targetInput.placeholder = "0"; } catch(e) { if(targetInput) targetInput.placeholder = "0"; } }
